<script>

  // ===== 共通ユーティリティ =====

  function dateStrLocal(d){

    const y=d.getFullYear(), m=('0'+(d.getMonth()+1)).slice(-2), day=('0'+d.getDate()).slice(-2);

    return `${y}-${m}-${day}`;

  }


  // ===== 保存キー =====

  const KEY='timebank_v7';

  const MASCOT9_KEY='timebank_mascots9_v1';

  const CYCLE_KEY='timebank_cycle_idx_v1'; // 次に使うスロット番号 0..8


  // ===== 状態読み込み／保存 =====

  const state = JSON.parse(localStorage.getItem(KEY) || '{"byDate":{}}');

  function persist(){

    localStorage.setItem(KEY, JSON.stringify(state));

    renderAll();

  }


  let mascots9 = JSON.parse(localStorage.getItem(MASCOT9_KEY) || '["","","","","","","","",""]');

  function persistMascots(){

    localStorage.setItem(MASCOT9_KEY, JSON.stringify(mascots9));

    renderAll();

  }


  let cycleIdx = parseInt(localStorage.getItem(CYCLE_KEY)||'0',10) || 0; // Saveのたびに +1 (mod 9)


  // ===== 参照 =====

  let viewMonth = new Date();

  let selectedDate = dateStrLocal(new Date());

  const sheetCount=document.getElementById('sheetCount');

  const spentMin=document.getElementById('spentMin');

  const useMin=document.getElementById('useMin');

  const previewBase=document.getElementById('previewBase');

  const previewSave=document.getElementById('previewSave');

  const selectedLabel=document.getElementById('selectedLabel');

  const totalMinutesEl=document.getElementById('totalMinutes');

  const mascotEl=document.getElementById('mascot');

  const orderHint=document.getElementById('orderHint');


  const calEl=document.getElementById('calendar');

  const monthLabel=document.getElementById('monthLabel');


  // 未設定スロット用の絵文字

  const fallback9=['🐰','🐱','🦄','🐻','🐼','🐨','🐯','🐸','👑'];


  // ===== 入力プレビュー =====

  function computeSaved(){

    const cnt=Math.max(1,parseInt(sheetCount.value||'1',10));

    const spent=Math.max(0,parseInt(spentMin.value||'0',10));

    return Math.max(0, 15*cnt - spent);

  }

  function updateBasePreview(){

    const cnt=Math.max(1,parseInt(sheetCount.value||'1',10));

    previewBase.textContent=`15 × ${cnt} = ${15*cnt}分`;

  }

  function updateSavePreview(){

    previewSave.textContent=`今日の貯金：+${computeSaved()}分`;

  }

  sheetCount.addEventListener('input', ()=>{ updateBasePreview(); updateSavePreview(); });

  spentMin.addEventListener('input', updateSavePreview);

  updateBasePreview(); updateSavePreview();


  // ===== 残高 =====

  function totalBalance(){

    let s=0, b=state.byDate||{};

    for(const k in b){ s+=(b[k].save||0)-(b[k].spend||0); }

    return s;

  }


  // ===== マスコット表示（現在のサイクルに基づく）=====

  function updateMascotByCycle(){

    // 直近表示は「前回使われたスロット」。初期表示では (cycleIdx+8)%9 を見せる

    const idx=(cycleIdx+8)%9;

    const url=mascots9[idx];

    mascotEl.innerHTML = url ? `<img src="${url}" alt="mascot">` : fallback9[idx];

    orderHint.textContent = (cycleIdx%9)+1; // 次に使われる番号の表示

  }


  // ===== 効果音 =====

  function sparkle(){

    const ctx=new (window.AudioContext||window.webkitAudioContext)();

    const now=ctx.currentTime;

    const osc=ctx.createOscillator(); const gain=ctx.createGain();

    osc.type='triangle';

    osc.frequency.setValueAtTime(880,now);

    osc.frequency.exponentialRampToValueAtTime(1760,now+0.18);

    gain.gain.setValueAtTime(0.0001,now);

    gain.gain.exponentialRampToValueAtTime(0.08,now+0.02);

    gain.gain.exponentialRampToValueAtTime(0.0001,now+0.22);

    osc.connect(gain).connect(ctx.destination);

    osc.start(now); osc.stop(now+0.25);

  }


  // ===== 記録（各日、save/spendは最後の1件のみ保持）=====

  function setRecord(dateStr,type,minutes){

    if(!state.byDate[dateStr]) state.byDate[dateStr]={save:0,spend:0};

    state.byDate[dateStr][type]=Math.max(0, Math.floor(Number(minutes)||0));

    persist();

  }


  // ===== ボタン動作 =====

  document.getElementById('saveBtn').onclick=()=>{

    const saved=computeSaved();

    setRecord(selectedDate,'save',saved);

    sparkle();


    // マスコットを次スロットに切替

    const used=cycleIdx;

    const url=mascots9[used];

    mascotEl.innerHTML = url ? `<img src="${url}" alt="mascot">` : fallback9[used];

    mascotEl.style.transform='scale(1.12)';

    setTimeout(()=>mascotEl.style.transform='',220);


    cycleIdx=(cycleIdx+1)%9;

    localStorage.setItem(CYCLE_KEY,String(cycleIdx));

    orderHint.textContent=(cycleIdx%9)+1;

  };


  document.getElementById('spendBtn').onclick=()=>{

    const mins=Math.max(1,parseInt(useMin.value||'1',10));

    setRecord(selectedDate,'spend',mins);

  };


  // ===== カレンダー描画 =====

  function renderCalendar(){

    const y=viewMonth.getFullYear(), m=viewMonth.getMonth();

    monthLabel.textContent=`${y}年 ${m+1}月`;

    calEl.innerHTML='';


    // 曜日ヘッダー

    ['日','月','火','水','木','金','土'].forEach(d=>{

      const h=document.createElement('div');

      h.className='dow'; h.textContent=d; calEl.appendChild(h);

    });


    const start=new Date(y,m,1), end=new Date(y,m+1,0);

    const startDow=start.getDay();


    // 1日の前の空白

    for(let i=0;i<startDow;i++){ calEl.appendChild(document.createElement('div')); }


    // 日付セル

    for(let i=1;i<=end.getDate();i++){

      const d=new Date(y,m,i);

      const ds=dateStrLocal(d);

      const cell=document.createElement('div');

      cell.className='day';

      if(ds===selectedDate) cell.classList.add('selected');


      const num=document.createElement('div'); num.className='num'; num.textContent=i;

      const rec=document.createElement('div'); rec.className='rec';


      const r=state.byDate[ds]; const parts=[];

      if(r){

        if(r.save>0) parts.push(`<span class="plus">+${r.save}分</span>`);

        if(r.spend>0) parts.push(`<span class="minus">-${r.spend}分</span>`);

      }

      rec.innerHTML=parts.join(' ');

      cell.appendChild(num); cell.appendChild(rec);


      cell.addEventListener('click', ()=>{

        selectedDate=ds;

        renderCalendar();

        selectedLabel.textContent=`選択中の日付：${ds}`;

      });


      calEl.appendChild(cell);

    }

  }


  document.getElementById('prevMonth').onclick=()=>{ viewMonth.setMonth(viewMonth.getMonth()-1); renderCalendar(); };

  document.getElementById('nextMonth').onclick=()=>{ viewMonth.setMonth(viewMonth.getMonth()+1); renderCalendar(); };


  // ===== 全体再描画 =====

  function renderAll(){

    totalMinutesEl.textContent = totalBalance() + '分';

    selectedLabel.textContent = selectedDate ? `選択中の日付：${selectedDate}` : '日付をカレンダーから選んでね';

    renderCalendar();

    updateBasePreview(); updateSavePreview();

    updateMascotByCycle();

  }

  renderAll();


  // ===== 設定モーダル =====

  const modal=document.getElementById('settingsModal');

  const openBtn=document.getElementById('openSettings');

  const closeBtn=document.getElementById('closeSettings');

  const clearBtn=document.getElementById('clearMascots');

  const resetBtn=document.getElementById('resetCycle');

  const uploader=document.getElementById('uploader');


  function buildSlots(){

    uploader.innerHTML='';

    for(let i=0;i<9;i++){

      const slot=document.createElement('div'); slot.className='slot';

      slot.innerHTML = `

        <h5>${i+1}番（順番）</h5>

        <div class="thumb" id="thumb${i}">${mascots9[i] ? `<img src="${mascots9[i]}" alt="m${i}">` : '<span>＋</span>'}</div>

        <input id="file${i}" type="file" accept="image/*">

      `;

      uploader.appendChild(slot);

    }

    // ハンドラ付与

    for(let i=0;i<9;i++){

      const inp=document.getElementById('file'+i);

      inp.addEventListener('change', ()=>{

        const file=inp.files && inp.files[0]; if(!file) return;

        const reader=new FileReader();

        reader.onload=e=>{ mascots9[i]=e.target.result; persistMascots(); buildSlots(); };

        reader.readAsDataURL(file);

      });

    }

  }

  buildSlots();


  openBtn.onclick=()=>{ modal.style.display='flex'; modal.setAttribute('aria-hidden','false'); };

  closeBtn.onclick=()=>{ modal.style.display='none'; modal.setAttribute('aria-hidden','true'); };

  modal.addEventListener('click', (e)=>{ if(e.target===modal){ closeBtn.click(); }});

  clearBtn.onclick=()=>{ mascots9=["","","","","","","","",""]; persistMascots(); buildSlots(); };

  resetBtn.onclick=()=>{ cycleIdx=0; localStorage.setItem(CYCLE_KEY,'0'); updateMascotByCycle(); };

</script>
